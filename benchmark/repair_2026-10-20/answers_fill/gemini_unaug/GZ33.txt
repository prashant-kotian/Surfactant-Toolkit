def solve_gz33():

    # Dodecyl tail: C12H25-
    # Polyether chain: O-(CH2-CH2-O)_n - H
    # Let's count "OCCO" in the string? Wait, it's O C C O C C O...
    # The string after dodecyl is OCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCOCCO
    tail = "CCCCCCCCCCCC"
    head = smiles[len(tail):]
    # head is OCCOCCO...
    # "OCC" is one unit if we consider the terminal OH? 
    # Actually, a repeat unit is CH2-CH2-O (mass 44.05).
    # Let's count 'O's in the head.
    num_O = head.count('O')
    n_units = num_O  # since each EO unit has 1 oxygen. The terminal is OH.
    
    # Molecular weight:
    # Dodecyl = C12H25 = 12*12.011 + 25*1.008 = 144.132 + 25.2 = 169.332
    # Hydrophilic = (O-CH2-CH2)_n - H ? No, it's (CH2-CH2-O)_n - H. 
    # Let's look at the SMILES: C...C - O - C - C - O - C - C - O ...
    # Number of oxygens = n_units.
    # Number of carbons in head = head.count('C') = 2 * n_units
    # Hydrogen in head: each -CH2-CH2-O- is 4 H. The last one is -CH2-CH2-OH (so 5 H on the last unit? No, wait.)
    # Let's write the molecular formula. 
    # Tail: C12H25
    # Head attached: (O-CH2-CH2)_n - OH ? 
    # Let's count carefully from SMILES: C(12) - O - C - C - O - C - C ...
    # If n_units = 2, it would be C-C-O-C-C-O. SMILES: CCOCCO.
    # Wait, the SMILES ends with O. But SMILES implies valid valences. 
    # If it ends in O, it's an alcohol (-OH).
    
    print("GZ-33:")
    print("num O:", num_O)
    print("num C in head:", head.count('C'))
    
    # Mass of tail C12H25 = 12 * 12.011 + 25 * 1.008
    m_tail = 12 * 12.011 + 25 * 1.008
    
    # Mass of head: num_O * O + (num_C) * C + (num_C * 2 + 1) * H 
    # Since each carbon is a CH2, and the terminal oxygen has an H.
    m_head = num_O * 15.999 + head.count('C') * 12.011 + (head.count('C') * 2 + 1) * 1.008
    m_total = m_tail + m_head
    hlb = 20 * m_head / m_total
    
    print(f"n_units: {num_O}")
    print(f"m_tail: {m_tail}")
    print(f"m_head: {m_head}")
    print(f"m_total: {m_total}")
    print(f"hlb: {hlb}")
